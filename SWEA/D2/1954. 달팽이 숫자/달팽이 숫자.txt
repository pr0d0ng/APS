T = int(input())

for tc in range(1, T+1):
    N = int(input())
    total_num = N * N

    arr = list([0] * N for _ in range(N))
    
    i, j = 0, 0
    d = 0
    di = [0, 1, 0, -1]
    dj = [1, 0, -1, 0]

    for num in range(1, total_num + 1):
        arr[i][j] = num

        ni = i + di[d]
        nj = j + dj[d]

        if ni < 0 or ni >= N or nj < 0 or nj >= N or arr[ni][nj] != 0:
            d = (d+1) % 4
            ni = i + di[d]
            nj = j + dj[d]

        i, j = ni, nj

    print(f"#{tc}")
    for row in arr:
        print(*row)

import heapq

T = int(input())

for tc in range(1, T+1):
    N, E = map(int, input().split())

    graph = [[] for _ in range(N + 1)]
    for _ in range(E):
        s, e, w = map(int, input().split())
        graph[s].append((e, w))

    INF = float('inf')
    dist = [INF] * (N+1)
    dist[0] = 0

    pq = [(0, 0)]

    while pq:
        cur_dist, u = heapq.heappop(pq)

        if dist[u] < cur_dist:
            continue

        for v, w in graph[u]:
            next_dist = cur_dist + w

            if next_dist < dist[v]:
                dist[v] = next_dist
                heapq.heappush(pq, (next_dist, v))

    print(f"#{tc} {dist[N]}")
